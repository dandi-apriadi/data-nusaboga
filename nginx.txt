## Nginx config for lyviawarisanrasa.com (preserve /api and serve SPA + uploads)
## Revised: align with SPA build, API at :5000, websocket upgrades, caching, and basic security headers

server {
	server_name lyviawarisanrasa.com www.lyviawarisanrasa.com;

	# Allow larger uploads (payment proof, etc.)
	client_max_body_size 10m;

	# Serve React build (CRA output)
	root /var/www/nusaboga/frontend/build;
	index index.html index.htm;

	# Serve static uploads directly (bypass Node.js for images/files)
	# Use ^~ so this block wins over regex image handlers below
	location ^~ /uploads/ {
		alias /var/www/nusaboga/backend/public/uploads/;
		expires 1M;
		access_log off;
		add_header Cache-Control "public";
	}

	# SPA fallback
	location / {
		try_files $uri $uri/ /index.html;
	}

	# API proxy - preserve /api (and /api/v1) prefix to match Express mounts
	# IMPORTANT: trailing slash on location, NO trailing slash on proxy_pass.
	location /api/ {
		proxy_pass http://localhost:5000;
		proxy_http_version 1.1;
		proxy_set_header Host $host;
		proxy_set_header X-Forwarded-Host $host;
		proxy_set_header X-Real-IP $remote_addr;
		proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
		proxy_set_header X-Forwarded-Proto $scheme;
		proxy_set_header X-Forwarded-Port $server_port;
		proxy_redirect off;
		proxy_read_timeout 300s;
	}

	# Socket.IO / WebSocket upgrades (use ^~ to prefer this over regex)
	location ^~ /socket.io/ {
		proxy_pass http://localhost:5000;
		proxy_http_version 1.1;
		proxy_set_header Upgrade $http_upgrade;
		proxy_set_header Connection "upgrade";
		proxy_set_header Host $host;
		proxy_set_header X-Forwarded-Host $host;
		proxy_set_header X-Real-IP $remote_addr;
		proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
		proxy_set_header X-Forwarded-Proto $scheme;
		proxy_set_header X-Forwarded-Port $server_port;
		proxy_cache_bypass $http_upgrade;
		proxy_read_timeout 300s;
	}

	# Optional: cache static assets from build for better performance
	location ~* \.(?:css|js|svg|gif|png|jpg|jpeg|webp|ico)$ {
		try_files $uri =404;
		expires 7d;
		access_log off;
		add_header Cache-Control "public, must-revalidate, proxy-revalidate";
	}

	# Basic security headers (safe defaults)
	add_header X-Frame-Options "SAMEORIGIN" always;
	add_header X-Content-Type-Options "nosniff" always;
	add_header Referrer-Policy "strict-origin-when-cross-origin" always;
	add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

	# gzip compression (can also be placed at http { } level)
	gzip on;
	gzip_comp_level 5;
	gzip_min_length 1024;
	gzip_types text/plain text/css application/json application/javascript application/xml+rss image/svg+xml;

	listen [::]:443 ssl http2 ipv6only=on; # managed by Certbot
	listen 443 ssl http2; # managed by Certbot
	ssl_certificate /etc/letsencrypt/live/lyviawarisanrasa.com/fullchain.pem; # managed by Certbot
	ssl_certificate_key /etc/letsencrypt/live/lyviawarisanrasa.com/privkey.pem; # managed by Certbot
	include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
	ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}


server {
	listen 80;
	listen [::]:80;
	server_name lyviawarisanrasa.com www.lyviawarisanrasa.com;
	# Redirect all HTTP to HTTPS
	return 301 https://$host$request_uri;
}
